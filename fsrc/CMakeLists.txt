pkg_check_modules(NEP_NETCDFF REQUIRED IMPORTED_TARGET netcdf-fortran>=4.6.2)
include(CheckFortranSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES PkgConfig::NEP_NETCDFF)
set(CMAKE_REQUIRED_INCLUDES ${NEP_NETCDFF_INCLUDE_DIRS})
check_fortran_source_compiles("program probe
use iso_c_binding
use netcdf
integer, parameter :: valid = 1 / merge(1,0,kind(1)==c_int)
integer :: s
s = nf90_close(-1)
end program" NEP_NETCDFF_WORKS SRC_EXT F90)
unset(CMAKE_REQUIRED_LIBRARIES)
unset(CMAKE_REQUIRED_INCLUDES)
if(NOT NEP_NETCDFF_WORKS)
    message(FATAL_ERROR "NEXTCDF-4 Fortran requires a compatible netcdf.mod, library and default INTEGER matching C int")
endif()

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS small_float_procedure.F90.in)
set(PROCEDURES "")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/small_float_procedure.F90.in" TEMPLATE)
foreach(TYPE float16 bfloat16)
    foreach(OP get put)
        foreach(RANK RANGE 0 7)
            set(DIMS "")
            set(SLICE "")
            if(RANK EQUAL 0)
                set(SHAPE "[integer(c_size_t)::]")
                set(PACK "buffer(1) = values")
                set(UNPACK "if (status == nf90_noerr .or. status == nf90_erange) values = buffer(1)")
            else()
                foreach(I RANGE 1 ${RANK})
                    if(I GREATER 1)
                        string(APPEND DIMS ",")
                        string(APPEND SLICE ",")
                    endif()
                    string(APPEND DIMS ":")
                    string(APPEND SLICE "1:edges(${I})")
                endforeach()
                set(DIMS "(${DIMS})")
                set(SHAPE "shape(values,kind=c_size_t)")
                set(PACK "buffer = reshape(values(${SLICE}),[n])")
                set(UNPACK "if (status == nf90_noerr .or. status == nf90_erange) &\n      values(${SLICE}) = reshape(buffer,edges(1:${RANK}))")
            endif()
            if(OP STREQUAL "put")
                set(INTENT "in")
                set(WRITE 1)
                set(UNPACK "")
            else()
                set(INTENT "inout")
                set(WRITE 0)
                set(PACK "buffer = 0")
            endif()
            string(CONFIGURE "${TEMPLATE}" PROCEDURE @ONLY)
            string(APPEND PROCEDURES "${PROCEDURE}\n")
        endforeach()
    endforeach()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/procedures.inc" "${PROCEDURES}")
add_library(nepnextcdf4f SHARED nep_nextcdf4.F90 nep_nextcdf4_c.c)
set_target_properties(nepnextcdf4f PROPERTIES
    Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules"
    VERSION ${PROJECT_VERSION} SOVERSION 1
    INSTALL_RPATH "$ORIGIN;${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}"
    INSTALL_RPATH_USE_LINK_PATH TRUE)
target_include_directories(nepnextcdf4f PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_SOURCE_DIR}/include"
    PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/modules>"
           "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/nep>")
target_link_libraries(nepnextcdf4f PUBLIC PkgConfig::NEP_NETCDFF PRIVATE ncnextcdf4 nep)
install(TARGETS nepnextcdf4f EXPORT NEPTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/modules/nep_nextcdf4.mod" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/nep")
set(NEP_HAVE_NEXTCDF4_FORTRAN ON PARENT_SCOPE)
